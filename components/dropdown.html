// Abrir / cerrar el dropdown
function toggleDropdown(triggerId, dropdownId) {
  const dropdown = document.getElementById(dropdownId);
  const trigger  = document.getElementById(triggerId);
  if (!dropdown) return;
  const isOpen = !dropdown.hidden;
  dropdown.hidden = isOpen;
  if (trigger) trigger.setAttribute('aria-expanded', String(!isOpen));
}

// Seleccionar / deseleccionar un ítem
function selectDropdownItem(el) {
  if (el.classList.contains('dropdown-item--disabled')) return;
  const isActive = el.classList.contains('dropdown-item--active');
  const icon = el.querySelector('.dropdown-item__checkbox .material-icons-outlined');
  if (isActive) {
    el.classList.remove('dropdown-item--active');
    if (icon) icon.textContent = 'check_box_outline_blank';
  } else {
    el.classList.add('dropdown-item--active');
    if (icon) icon.textContent = 'check_box';
  }
}

// Cerrar al hacer clic fuera
document.addEventListener('click', e => {
  document.querySelectorAll('.dropdown:not([hidden])').forEach(dropdown => {
    const trigger = document.querySelector(`[aria-controls="${dropdown.id}"]`);
    if (!trigger) return; // sin trigger = preview estático, no cerrar
    if (!dropdown.contains(e.target) && !trigger.contains(e.target)) {
      dropdown.hidden = true;
      trigger.setAttribute('aria-expanded', 'false');
    }
  });
});

Estructura

BloqueClaseUso
Dropdown completo.dropdownEs la base del componente: contenedor + lista + footer opcional. Para armar uno propio, copiá el CSS, HTML y JS de esta sección y reemplazá los ítems de .dropdown__list por la variante de "List options" que necesites. Se pueden mezclar variantes dentro de un mismo dropdown.

List options

Cada variante es independiente entre sí

VarianteClaseUso
Default.dropdown-itemEs el ítem más simple, sin checkbox ni adornos extra.
Default + Number.dropdown-item__numberSuma un número a la derecha. Se usa para mostrar cantidades.
Default + Desc.dropdown-item__descSuma una línea de descripción debajo del título, truncada a 2 líneas.
Default + Pill.dropdown-item__statusSuma una pill de estado a la derecha. La pill siempre es read-only dentro del ítem.
Checkbox.dropdown-item__checkboxAgrega un checkbox a la izquierda. Se usa cuando el dropdown permite selección múltiple.
Checkbox + Number.dropdown-item__checkboxCombina el checkbox de selección múltiple con un número a la derecha.
Checkbox + Pill.dropdown-item__checkboxCombina el checkbox de selección múltiple con una pill de estado a la derecha.
Loading.dropdown-item--loadingReemplaza todos los ítems mientras se cargan los resultados.
No matches.dropdown-item--no-matchesReemplaza todos los ítems cuando la búsqueda no tiene resultados.
Group title.dropdown-item__group-titleEncabezado de grupo, no interactivo. Separa secciones dentro de la misma lista, con descripción opcional.
Section title.dropdown-item__section-titleSubencabezado dentro de un grupo. Admite un checkbox opcional (ej. "Seleccionar todos") o, con el modificador --desc, una descripción en su lugar.
